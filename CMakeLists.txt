cmake_minimum_required(VERSION 3.20)
project(Framoryx VERSION 0.3.0 LANGUAGES C CXX)
option(FRAMORYX_BUILD_DESKTOP "Build the shared Windows/Linux Qt desktop application" OFF)
enable_testing()
add_executable(FramoryxCoreTests source/tests.cpp)
target_compile_features(FramoryxCoreTests PRIVATE cxx_std_20)
add_test(NAME shared-core COMMAND FramoryxCoreTests)
add_executable(FramoryxMalformedInputTests source/fuzz_tests.cpp)
target_compile_features(FramoryxMalformedInputTests PRIVATE cxx_std_20)
add_test(NAME malformed-input COMMAND FramoryxMalformedInputTests)
add_executable(FramoryxDependencyTests source/dependency_tests.cpp source/vendor/imgui/imgui.cpp source/vendor/imgui/imgui_draw.cpp source/vendor/imgui/imgui_tables.cpp source/vendor/imgui/imgui_widgets.cpp)
target_compile_features(FramoryxDependencyTests PRIVATE cxx_std_20)
if(WIN32)
  target_link_libraries(FramoryxDependencyTests PRIVATE imm32)
endif()
add_test(NAME dependency-regressions COMMAND FramoryxDependencyTests)
add_executable(FramoryxFontBitmapTests source/font_bitmap_tests.cpp)
target_compile_features(FramoryxFontBitmapTests PRIVATE cxx_std_20)
add_test(NAME font-bitmap-bounds COMMAND FramoryxFontBitmapTests)
add_executable(FramoryxTelemetryStreamTests runtime/mangohud/stream_tests.cpp)
target_compile_features(FramoryxTelemetryStreamTests PRIVATE cxx_std_20)
add_test(NAME telemetry-stream COMMAND FramoryxTelemetryStreamTests)
add_executable(FramoryxHudPacketTests runtime/mangohud/hud_tests.cpp)
target_compile_features(FramoryxHudPacketTests PRIVATE cxx_std_17)
target_include_directories(FramoryxHudPacketTests PRIVATE source)
add_test(NAME game-hud-packets COMMAND FramoryxHudPacketTests)
if(FRAMORYX_BUILD_DESKTOP)
  add_subdirectory(desktop)
endif()
